% Pandoc template for the print interior.
% Page size = trim + bleed on every side (e.g. 128x188 trim + 5mm => 138x198).
% All margins in book/layout.yaml are measured from the trim line.
% Language settings (fonts, labels, `hangul`) come from <lang>/metadata.yaml.
\documentclass[10pt,twoside,openany]{book}

% ---- page geometry -------------------------------------------------------
\newlength\trimW  \setlength\trimW{$trim-width$mm}
\newlength\trimH  \setlength\trimH{$trim-height$mm}
\newlength\bleed  \setlength\bleed{$bleed$mm}
\usepackage[
  paperwidth=\dimexpr\trimW+2\bleed\relax,
  paperheight=\dimexpr\trimH+2\bleed\relax,
  top=\dimexpr$margin-top$mm+\bleed\relax,
  bottom=\dimexpr$margin-bottom$mm+\bleed\relax,
  inner=\dimexpr$margin-inner$mm+\bleed\relax,
  outer=\dimexpr$margin-outer$mm+\bleed\relax,
  headheight=12pt, headsep=7mm, footskip=10mm,
]{geometry}

\makeatletter
% Mark the cut line in the PDF (TrimBox) and the full page as BleedBox.
% Print shops and preflight tools (e.g. Acrobat) read these boxes.
\AddToHook{shipout/background}{%
  \put(0,0){\special{pdf:put @thispage <<
    /TrimBox [\strip@pt\dimexpr\bleed*7200/7227\relax\space
              \strip@pt\dimexpr\bleed*7200/7227\relax\space
              \strip@pt\dimexpr(\trimW+\bleed)*7200/7227\relax\space
              \strip@pt\dimexpr(\trimH+\bleed)*7200/7227\relax]
    /BleedBox [0 0
              \strip@pt\dimexpr(\trimW+2\bleed)*7200/7227\relax\space
              \strip@pt\dimexpr(\trimH+2\bleed)*7200/7227\relax]
  >>}}}
\makeatother

\makeatletter

\newcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}\fi}
