% Book cover: back + spine + front in one page, with bleed on every side.
% Page 1 = outside (back | spine | front), page 2 = inside of the cover
% (inside front | spine | inside back), for covers printed on both sides.
% Matches the RedPrinting SoftCover guide:
%   front 128x188, back 128x188, spine = \spine mm, bleed 5mm each side
%   -> page = (128 + spine + 128 + 10) x 198 mm
% The panel size can be overridden like the spine (`\def\trimW{148}
% \def\trimH{210}` for the A5 edition, see `make cover-a5`).
%
% Build with the Makefile: `make cover SPINE=1.36` (`make guides` adds
% cut/spine/safe-area lines to check placement against the template).
%
% `\def\split{}` (see `make cover-split`) drops the spine and puts the
% front and back covers on separate pages instead:
%   page 1 = front, page 2 = back, each (128 + 10) x (188 + 10) mm.
\providecommand\spine{7.85}

\documentclass{article}
\providecommand\trimW{128}   % mm, one panel
\providecommand\trimH{188}
\newcommand\bleed{5}
\newcommand\safe{8}      % keep text at least this far inside the cut line

\newcommand\booktitle{프로덕트를 처음부터 디자인하는 방법}
\newcommand\bookauthor{박근우}

\usepackage{xfp}
\ifdefined\split
  \edef\pageW{\fpeval{\trimW + 2*\bleed}}
\else
  \edef\pageW{\fpeval{2*\trimW + \spine + 2*\bleed}}
\fi
\edef\pageH{\fpeval{\trimH + 2*\bleed}}
% x positions (mm from the left edge of the file)
\edef\spineL{\fpeval{\bleed + \trimW}}            % back cover | spine
\edef\spineR{\fpeval{\bleed + \trimW + \spine}}   % spine | front cover
\edef\frontR{\fpeval{\spineR + \trimW}}           % front cover cut line
\edef\topY{\fpeval{\bleed + \trimH}}

\usepackage[paperwidth=\pageW mm, paperheight=\pageH mm, margin=0pt]{geometry}
\usepackage{fontspec}
\usepackage[hangul]{xetexko}
\setmainfont{Noto Serif CJK KR}\setmainhangulfont{Noto Serif CJK KR}
\setsansfont{Noto Sans CJK KR}\setsanshangulfont{Noto Sans CJK KR}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}

% Print colours in CMYK.
\selectcolormodel{cmyk}
\definecolor{bg}{cmyk}{0.02,0.10,0.85,0}      % warm yellow (like the sketch photo)
\definecolor{ink}{cmyk}{0,0,0,1}              % pure black (K only)
\definecolor{ink2}{cmyk}{0,0,0,0.75}

% TrimBox = cut line, BleedBox = whole page (read by print preflight).
\AddToHook{shipout/background}{\put(0,0){\special{pdf:put @thispage <<
  /TrimBox [\fpeval{\bleed*72/25.4} \fpeval{\bleed*72/25.4}
            \fpeval{(\pageW-\bleed)*72/25.4} \fpeval{(\pageH-\bleed)*72/25.4}]
  /BleedBox [0 0 \fpeval{\pageW*72/25.4} \fpeval{\pageH*72/25.4}] >>}}}

% Background of the inside of the cover (page 2). Use {cmyk}{0,0,0,0} for
% unprinted white paper.
\definecolor{insidebg}{cmyk}{0,0,0,0}  % white (no ink); use \colorlet{insidebg}{bg} for yellow

% Cut / safe-area lines for one panel on its own page (split mode).
\newcommand\panelguides{%
  \ifdefined\guides
    \begin{scope}[line width=0.25pt]
      \draw[red] (\bleed,\bleed) rectangle (\bleed+\trimW,\topY);
      \draw[blue, dashed] (\bleed+\safe,\bleed+\safe)
                          rectangle (\bleed+\trimW-\safe,\topY-\safe);
    \end{scope}
  \fi}

% Cut / spine / safe-area lines, drawn only with `make guides`.
\newcommand\guidelines{%
  \ifdefined\guides
    \begin{scope}[line width=0.25pt]
      \draw[red] (\bleed,\bleed) rectangle (\frontR,\topY);           % cut
      \draw[red] (\spineL,0) -- (\spineL,\pageH) (\spineR,0) -- (\spineR,\pageH);
      \draw[blue, dashed] (\bleed+\safe,\bleed+\safe)
                          rectangle (\spineL-\safe,\topY-\safe);       % safe area
      \draw[blue, dashed] (\spineR+\safe,\bleed+\safe)
                          rectangle (\frontR-\safe,\topY-\safe);
    \end{scope}
  \fi}

% Panel contents, drawn from the bottom-left corner of the cut line.
\newcommand\frontpanel{%
    \node[anchor=north west, align=left, inner sep=0, text=ink,
          font=\sffamily\bfseries\fontsize{25}{34}\selectfont]
      at (14, \trimH-24) {프로덕트를\\처음부터\\디자인하는 방법};

    % design -> build -> evaluate cycle (same as figures/design-cycle.tex),
    % centred across the panel and in the space between title and author
    \begin{scope}[shift={(\trimW/2,\trimH/2-29.5)}, line width=0.9pt, draw=ink,
                  every node/.style={text=ink}]
      \def\R{26}
      \foreach \a/\t in {90/디자인, -30/구현, 210/평가}
        \draw (\a:\R) circle (10)
          node[font=\sffamily\bfseries\fontsize{9}{11}\selectfont] {\t};
      \foreach \s/\e in {64/-4, -56/-124, 184/116}
        \draw[-{Stealth[length=2.6mm,width=2mm]}]
          (\s:\R) arc[start angle=\s, end angle=\e, radius=\R];
      \node[align=center, font=\sffamily\fontsize{7.5}{9.5}\selectfont]
        at (0,0) {끝없는\\반복};
    \end{scope}

    \node[anchor=south west, inner sep=0, text=ink,
          font=\sffamily\fontsize{11}{14}\selectfont]
      at (14, 16) {\bookauthor};
}
\newcommand\backpanel{%
    \node[anchor=north west, text width=\fpeval{\trimW-32}mm, align=justify, inner sep=0,
          text=ink, font=\fontsize{8.5}{14.5}\selectfont]
      at (16, \trimH-28) {%
        이 소책자는 사용자 중심으로 제품을 디자인하는 방법을 설명한다.
        상황 파악에서 시작해 니즈 도출, 스케치, 프로토타이핑,
        그리고 디자인 커뮤니케이션과 리포트 작성까지,
        챕터를 따라가면 제품 하나를 디자인할 수 있도록 구성했다.
        마지막으로 AI로 인해 변하는 것과 변하지 않는 것을 이야기한다.\par\medskip
        \textsf{\bfseries 이런 분들에게}\par\smallskip
        · UX 디자이너로 일을 하고 싶은 학생 또는 디자이너\par
        · 구현뿐만 아니라 서비스 설계까지 하고 싶은 개발자\par
        · 사용자 중심으로 제품을 만들고 싶은 모든 사람};
    \node[anchor=south west, inner sep=0, text=ink2,
          font=\sffamily\fontsize{7.5}{9}\selectfont]
      at (16, 16) {\booktitle\quad\bookauthor};
}

  \end{scope}
  \panelguides
\end{tikzpicture}

    shift={(current page.south west)}]

  % background: fill the whole file, including the 5mm bleed
  \fill[bg] (0,0) rectangle (\pageW,\pageH);

  % ---- front cover ------------------------------------------------------
  \begin{scope}[shift={(\spineR,\bleed)}]
    \frontpanel
  \end{scope}

  % ---- back cover -------------------------------------------------------
  \begin{scope}[shift={(\bleed,\bleed)}]
    \backpanel
  \end{scope}

  % ---- spine ------------------------------------------------------------
  % Text only fits on a spine of about 5mm or more.
  \ifdim\spine pt<5pt\else
    \node[rotate=-90, text=ink, inner sep=0,
          font=\sffamily\bfseries\fontsize{\fpeval{min(9,\spine*1.6)}}{10}\selectfont]
      at ({(\spineL+\spineR)/2}, {\bleed+\trimH/2})
      {\booktitle\hspace{8mm}\mdseries\bookauthor};
  \fi

  \guidelines
\end{tikzpicture}

\newpage

  \begin{scope}[shift={(\bleed,\bleed)}]   % inside front
  \end{scope}
  \begin{scope}[shift={(\spineR,\bleed)}]  % inside back
  \end{scope}

  \guidelines
\end{tikzpicture}
\fi
