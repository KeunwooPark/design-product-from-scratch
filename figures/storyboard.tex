  \def\W{22mm}\def\H{22mm}

    \draw[hand] (#1,#2+6.7mm) -- (#1,#2+3mm);
    \draw[hand] (#1,#2+3mm) -- (#1-1.3mm,#2) (#1,#2+3mm) -- (#1+1.3mm,#2);
  }
  % small phone: #1 x, #2 y (center)
  \def\phone#1#2{\draw[hthin, fill=white] (#1-0.9mm,#2-1.5mm) rectangle (#1+0.9mm,#2+1.5mm);}

  % ---------- 1. scan the QR code at the entrance ----------
  \begin{scope}[shift={(0,0)}]
    \draw[hand] (0,0) rectangle (\W,\H);
    \node[font=\figtiny\bfseries, anchor=north west, inner sep=1mm] at (0,\H) {1};
    \draw[hthin] (0,2mm) -- (\W,2mm);

    \draw[hand] (12mm,2mm) rectangle (20mm,18mm);
    \node[font=\figtiny] at (16mm,16mm) {경험커피};

    % sign under the QR code: after paying, wait inside (cut 3)
    \node[font=\fontsize{4.3}{5}\selectfont] at (16mm,6.1mm) {안쪽 대기};
    \draw[hthin, -{Stealth[length=1.2mm,width=1mm]}] (13.6mm,4.2mm) -- (18.6mm,4.2mm);
    \person{6mm}{2mm}
    \draw[hand] (6mm,7.2mm) -- (8.6mm,8.2mm);
    \phone{9.4mm}{8.6mm}

  \begin{scope}[shift={(24.3mm,0)}]
    \draw[hand] (0,0) rectangle (\W,\H);

    \draw[hand, fill=white] (6mm,2mm) rectangle (16mm,20mm);

      \node[font=\fontsize{4}{5}\selectfont] at (11mm,\y mm+0.1mm) {\t};
